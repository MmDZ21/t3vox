~~~kotlin
package com.toosarax.ts3client.protocol

import kotlinx.coroutines.Dispatchers
import kotlinx.coroutines.withContext
import org.xbill.DNS.Lookup
import org.xbill.DNS.SRVRecord
import org.xbill.DNS.Type

internal fun interface SrvRecordLookup {
    fun lookup(host: String): Pair<String, Int>?
}

internal object DnsJavaSrvRecordLookup : SrvRecordLookup {
    override fun lookup(host: String): Pair<String, Int>? {
        if (DnsJavaTs3ServerResolver.isIpv4Literal(host)) return null

        return runCatching {
            val lookup = Lookup("_ts3._udp.$host", Type.SRV)
            val answers = lookup.run()
            if (lookup.result != Lookup.SUCCESSFUL || answers.isNullOrEmpty()) {
                Ts3DebugLog.d(
                    "SRV lookup failed for _ts3._udp." + host +
                        " (result=" + lookup.result +
                        ", answers=" + (answers?.size ?: 0) + ")",
                )
                return null
            }
            val record = answers.firstOrNull { it is SRVRecord } as? SRVRecord
            if (record == null) {
                Ts3DebugLog.d("SRV lookup for _ts3._udp.$host returned no SRV records")
                return null
            }
            val target = record.target.toString(true).removeSuffix(".")
            if (target.isBlank() || record.port !in 1..65535) return null
            Ts3DebugLog.d(
                "SRV lookup _ts3._udp." + host + " -> " + target + ":" + record.port,
            )
            target to record.port
        }.onFailure { error ->
            Ts3DebugLog.d("SRV lookup error for _ts3._udp." + host + ": " + error.message)
        }.getOrNull()
    }
}

class DnsJavaTs3ServerResolver internal constructor(
    private val srvLookup: SrvRecordLookup,
) : Ts3ServerResolver {
    constructor() : this(DnsJavaSrvRecordLookup)

    override suspend fun resolve(host: String, requestedPort: Int): ResolvedServerEndpoint =
        withContext(Dispatchers.IO) {
            val normalizedHost = normalizeHost(host)
            if (requestedPort != DEFAULT_TS3_PORT) {
                return@withContext ResolvedServerEndpoint(
                    host = normalizedHost,
                    port = requestedPort,
                )
            }

            val srv = srvLookup.lookup(normalizedHost)
            if (srv != null) {
                ResolvedServerEndpoint(
                    host = srv.first,
                    port = srv.second,
                    srvResolved = true,
                )
            } else {
                ResolvedServerEndpoint(
                    host = normalizedHost,
                    port = requestedPort,
                )
            }
        }

    companion object {
        fun normalizeHost(raw: String): String {
            var host = raw.trim()
            host = host.removePrefix("ts3server://")
            host = host.removePrefix("ts3://")
            if (host.contains("/")) {
                host = host.substringBefore("/")
            }
            if (host.contains(":") && !isIpv4Literal(host)) {
                host = host.substringBefore(":")
            }
            return host.trim().removeSuffix(".")
        }

        fun isIpv4Literal(host: String): Boolean {
            val parts = host.split(".")
            if (parts.size != 4) return false
            return parts.all { part ->
                part.toIntOrNull()?.let { value -> value in 0..255 } == true
            }
        }
    }
}
~~~

~~~powershell
[CmdletBinding()]
param(
    [string]$OutputPath = ""
)

Set-StrictMode -Version Latest
$ErrorActionPreference = "Stop"

$upstreamUrl = "https://jitpack.io/com/github/manevolent/ts3j/db57d60/ts3j-db57d60.jar"
$upstreamSha256 = "50515411094C5872FD8B62BDFB6C14AC6C5F6C5347CC4FD2A086327EFD7EC294"
$patchedSha256 = "D360466A61B92A131EF2CEF0CFC81D6DD5035FD23082B7E6D63CF69242C8E004"
$removedEntry = "com/github/manevolent/ts3j/protocol/packet/handler/client/LocalClientHandlerConnecting.class"

if ([string]::IsNullOrWhiteSpace($OutputPath)) {
    $OutputPath = Join-Path $PSScriptRoot "..\app\libs\ts3j-db57d60-patched.jar"
}

$temporaryDirectory = Join-Path ([IO.Path]::GetTempPath()) ("ts3j-" + [guid]::NewGuid().ToString("N"))
$upstreamJar = Join-Path $temporaryDirectory "ts3j-db57d60.jar"
$patchedJar = Join-Path $temporaryDirectory "ts3j-db57d60-patched.jar"
New-Item -ItemType Directory -Path $temporaryDirectory | Out-Null

try {
    Invoke-WebRequest -Uri $upstreamUrl -OutFile $upstreamJar -UseBasicParsing
    $downloadedHash = (Get-FileHash -Algorithm SHA256 $upstreamJar).Hash
    if ($downloadedHash -ne $upstreamSha256) {
        throw "Upstream SHA-256 mismatch. Expected $upstreamSha256, received $downloadedHash."
    }

    Copy-Item -LiteralPath $upstreamJar -Destination $patchedJar
    Add-Type -AssemblyName System.IO.Compression
    Add-Type -AssemblyName System.IO.Compression.FileSystem
    $archive = [System.IO.Compression.ZipFile]::Open(
        $patchedJar,
        [System.IO.Compression.ZipArchiveMode]::Update
    )
    try {
        $entry = $archive.GetEntry($removedEntry)
        if ($null -eq $entry) {
            throw "Expected JAR entry was not found: $removedEntry"
        }
        $entry.Delete()
    } finally {
        $archive.Dispose()
    }

    $resultHash = (Get-FileHash -Algorithm SHA256 $patchedJar).Hash
    if ($resultHash -ne $patchedSha256) {
        throw "Patched SHA-256 mismatch. Expected $patchedSha256, received $resultHash."
    }

    $outputDirectory = Split-Path -Parent $OutputPath
    New-Item -ItemType Directory -Path $outputDirectory -Force | Out-Null
    Copy-Item -LiteralPath $patchedJar -Destination $OutputPath -Force
    Write-Host "Rebuilt $OutputPath"
    Write-Host "SHA-256: $resultHash"
} finally {
    if (Test-Path -LiteralPath $temporaryDirectory) {
        Remove-Item -LiteralPath $temporaryDirectory -Recurse -Force
    }
}
~~~

- [x] **Step 1: Create scripts/check-publication-readiness.ps1**

~~~powershell
[CmdletBinding()]
param()

Set-StrictMode -Version Latest
$ErrorActionPreference = "Stop"
$repoRoot = (Resolve-Path (Join-Path $PSScriptRoot "..")).Path
Set-Location $repoRoot

$failures = [System.Collections.Generic.List[string]]::new()
$files = @(git ls-files --cached --others --exclude-standard)
$forbiddenFiles = @("local.properties", "live-debug-output.txt", "keystore.properties")

foreach ($file in $files) {
    $normalized = $file.Replace("\", "/")
    if ($forbiddenFiles -contains $normalized) {
        $failures.Add("Forbidden file is visible to Git: $file")
    }
    if ($normalized -match "(^|/)(app/build|build|\.android-sdk)/") {
        $failures.Add("Generated or SDK file is visible to Git: $file")
    }
    if ($normalized -match "\.(jks|keystore|pem|p12)$") {
        $failures.Add("Credential or signing file is visible to Git: $file")
    }
}

$textExtensions = @(
    ".java", ".json", ".kt", ".kts", ".md", ".properties",
    ".ps1", ".txt", ".xml", ".yaml", ".yml"
)
$forbiddenPatterns = [ordered]@{
    "personal Windows path" = "C:\\Users\\"
    "private test server" = ("bazitory" + "\.com")
    "private SRV target" = ("ts" + "\.hxts\.ir")
    "private debug nickname" = ("Cursor" + "Debug")
}

foreach ($file in $files) {
    if (-not (Test-Path -LiteralPath $file -PathType Leaf)) { continue }
    $extension = [IO.Path]::GetExtension($file).ToLowerInvariant()
    $isGitIgnore = [IO.Path]::GetFileName($file) -eq ".gitignore"
    if (-not $isGitIgnore -and $textExtensions -notcontains $extension) { continue }
    foreach ($entry in $forbiddenPatterns.GetEnumerator()) {
        if (Select-String -LiteralPath $file -Pattern $entry.Value -Quiet) {
            $failures.Add("$($entry.Key) found in $file")
        }
    }
}

$jarPath = "app/libs/ts3j-db57d60-patched.jar"
$expectedJarHash = "D360466A61B92A131EF2CEF0CFC81D6DD5035FD23082B7E6D63CF69242C8E004"
$removedEntry = "com/github/manevolent/ts3j/protocol/packet/handler/client/LocalClientHandlerConnecting.class"

if (-not (Test-Path -LiteralPath $jarPath)) {
    $failures.Add("Patched ts3j JAR is missing.")
} else {
    $actualJarHash = (Get-FileHash -Algorithm SHA256 $jarPath).Hash
    if ($actualJarHash -ne $expectedJarHash) {
        $failures.Add("Patched ts3j SHA-256 mismatch: $actualJarHash")
    }
    $jarEntries = @(jar tf $jarPath)
    if ($LASTEXITCODE -ne 0) {
        $failures.Add("Unable to inspect patched ts3j JAR with jar.")
    } elseif ($jarEntries -contains $removedEntry) {
        $failures.Add("Patched ts3j JAR still contains $removedEntry")
    }
}

if ($failures.Count -gt 0) {
    $failures | ForEach-Object { Write-Error $_ }
    exit 1
}
Write-Host "Publication readiness checks passed."
~~~
